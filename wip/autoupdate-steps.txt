Operation:
	Make the dependency tree.

			A
		B		C
	       DEF

	Start at root.
	Does it have an update?  If yes,
		Reset pkgrel. 
		Update pkgver.
		Build it.
	Otherwise, does it have a rebuilt dependency? If yes,
		Increment pkgrel.
		Build it.
	For each dependent, do the same check.

Pseudocode:
	struct node
	{
		int pkgrel;
		struct node* first_child;
		struct node* next_sibling;
	};
	struct node* build_deptree(...);
	int has_update(struct node* p);
	int makepkg(struct node* p);
	void handle_pkg(struct node* p, int rebuild)
	{
		int update;
		if (!p) {
			return;
		}
		update = has_update(p);
		if (update) {
			p->pkgrel = 0;
		}
		if (rebuild | update) {
			p->pkgrel += 1;
			makepkg(p);
		}
		handle_pkg(p->first_child, rebuild | update);
		handle_pkg(p->next_sibling, rebuild);
	}
	int main()
	{
		struct node* root = build_deptree(...);
		handle_pkg(root, 0);
	}

How to build the dependency tree?
	PKGBUILDs point at their dependencies. We need the dependents instead.
	Build the tree from PKGBUILDs, then reverse it.

	struct pkg
	{
		int pkgname;
		struct pkg** depends;
		struct pkg** makedepends;
	};
